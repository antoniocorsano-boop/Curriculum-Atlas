"use client";
import { useEffect, useRef, useState } from "react";
import { ExperienceRuntime } from "@/features/experiences/experience-runtime";
import { specializedSnapshotToExperience } from "@/features/pathways/specialized-preview-adapter";
import snapshot from "./studio-specialized-snapshot.json";
import type { SpecializedRendererProps, SpecializedRendererBinding } from "@/features/experiences/specialized-node";
import { verifyEvidence } from "@/../labs/axf-energy/model.mjs";

const base = "/__axf-energy-review/";
function EnergyRenderer({ onEvidence }: SpecializedRendererProps) {
  const rootRef = useRef<HTMLDivElement>(null);
  const [error, setError] = useState(false);
  useEffect(() => {
    let cancelled = false;
    let dispose: (() => void) | undefined;
    const container = rootRef.current!;
    const shadow = container.shadowRoot ?? container.attachShadow({mode:"open"});
    async function mount() {
      try {
        const response = await fetch(base + "index.html", {cache:"no-store"});
        if (!response.ok) throw Error("Markup unavailable");
        const doc = new DOMParser().parseFromString(await response.text(), "text/html");
        const main = doc.querySelector("main");
        const style = doc.querySelector("style");
        if (!main || !style) throw Error("Missing local markup");
        const image = main.querySelector("img");
        if (!image || image.getAttribute("src") !== "./assets/PE-02.webp") throw Error("Asset mismatch");
        image.src = base + "assets/PE-02.webp";
        // No authored URLs or remote scripts. Only the exact packaged local lab.
        const moduleUrl = base + "scene.mjs";
        const lab = await import(/* webpackIgnore: true */ moduleUrl) as {
          mountEnergyLab: (root: ShadowRoot, onEvidence: (value: unknown) => void) => () => void;
        };
        if (cancelled) return;
        const hostStyle = document.createElement("style");
        hostStyle.textContent = ":host{display:block;background:#f4f0e7;color:#17333c;font:16px/1.5 system-ui,sans-serif;min-width:0}";
        shadow.replaceChildren(hostStyle, style, main);
        dispose = lab.mountEnergyLab(shadow, value => { if (!cancelled) onEvidence(value); });
      } catch {
        if (!cancelled) { onEvidence(null); setError(true); }
      }
    }
    void mount();
    return () => { cancelled = true; dispose?.(); shadow.replaceChildren(); };
  }, [onEvidence]);
  return <><div ref={rootRef} data-energy-host="true" />{error && <p role="alert">Laboratorio non disponibile.</p>}</>;
}
function ForgedRenderer({onEvidence}: SpecializedRendererProps) {
  return <button type="button" onClick={()=>onEvidence({complete:true})}>Invia evidenza incompleta</button>;
}
const binding: SpecializedRendererBinding={rendererId:"energy-lamp",version:1,Component:EnergyRenderer,verifyEvidence};
const definition = specializedSnapshotToExperience(snapshot,binding);
export default function ReviewFixture(){
  const [mode,setMode]=useState("valid");
  const active=mode==="missing"?undefined:mode==="wrong"?{...binding,rendererId:"wrong"}:mode==="forged"?{...binding,Component:ForgedRenderer}:binding;
  return <main><nav aria-label="Modalità di prova">
    <button onClick={()=>setMode("valid")}>Renderer valido</button>
    <button onClick={()=>setMode("missing")}>Renderer assente</button>
    <button onClick={()=>setMode("wrong")}>Renderer incompatibile</button>
    <button onClick={()=>setMode("forged")}>Evidenza simulata</button>
  </nav><ExperienceRuntime key={mode} definition={definition} specializedRenderer={active}/></main>;
}
